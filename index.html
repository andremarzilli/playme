    <script>

        let audioCtx = null;

        let mainFilter = null;

        let delayNode = null;

        let delayFeedback = null;

        let reverbNode = null;

        let reverbGain = null;


        let currentWave = 'sine';

        let isPlaying = false;

        let currentStep = 0;

        let timerId = null;


        // Stato del Sequencer: 8 step. Ogni step ha: active (true/false), pitchStep (0-5 altezza), duration (0.1 - 0.8 secondi)

        let stepsData = Array.from({length: 8}, () => ({ active: false, pitchStep: 0, duration: 0.2 }));

        const pentatonicScale = [261.63, 293.66, 329.63, 392.00, 440.00, 523.25]; // Do, Re, Mi, Sol, La, Do


        // Costruisci graficamente il sequencer ad altezze

        const grid = document.getElementById('sequencer-grid');

        for(let i=0; i<8; i++) {

            let col = document.createElement('div');

            col.className = 'step-column';

            

            let bubble = document.createElement('div');

            bubble.className = 'kid-bubble';

            bubble.id = `bubble-${i}`;

            // Cambia altezza e attiva al click

            bubble.onclick = () => handleBubbleClick(i);

            

            let lenSlider = document.createElement('input');

            lenSlider.type = 'range';

            lenSlider.className = 'len-slider';

            lenSlider.min = '0.05'; lenSlider.max = '0.8'; lenSlider.step = '0.05'; lenSlider.value = '0.2';

            lenSlider.oninput = (e) => { stepsData[i].duration = parseFloat(e.target.value); };


            col.appendChild(bubble);

            col.appendChild(lenSlider);

            grid.appendChild(col);

        }


        function handleBubbleClick(index) {

            let data = stepsData[index];

            if (!data.active) {

                data.active = true;

                data.pitchStep = 0;

            } else {

                data.pitchStep = (data.pitchStep + 1) % 6;

                if (data.pitchStep === 0) {

                    data.active = false;

                }

            }

            renderSequencerUI();

            if (data.active && audioCtx) playPreviewTone(pentatonicScale[data.pitchStep], data.duration);

        }


        function renderSequencerUI() {

            for(let i=0; i<8; i++) {

                let bubble = document.getElementById(`bubble-${i}`);

                let data = stepsData[i];

                if(data.active) {

                    bubble.classList.add('active');

                    // Sposta fisicamente la bolla verso l'alto in base alla nota scelto (6 livelli)

                    bubble.style.bottom = `${(data.pitchStep * 22) + 25}px`;

                } else {

                    bubble.classList.remove('active');

                    bubble.style.bottom = '25px';

                }

            }

        }


        function setWave(wave, btn) {

            currentWave = wave;

            document.querySelectorAll('.wave-btn').forEach(b => b.classList.remove('active'));

            btn.classList.add('active');

            document.getElementById('display').innerText = `Voce cambiata in: ${btn.innerText}`;

        }


        function initAudioMachine() {

            audioCtx = new (window.AudioContext || window.webkitAudioContext)();

            

            mainFilter = audioCtx.createBiquadFilter();

            mainFilter.type = 'lowpass';


            delayNode = audioCtx.createDelay();

            delayFeedback = audioCtx.createGain();

            delayNode.delayTime.value = 0.25;


            // Il modulo di Riverbero Sintetico Spaziale

            reverbNode = audioCtx.createConvolver();

            reverbNode.buffer = createReverbImpulseResponse(2.5, 2.0); // 2.5 secondi di coda creata al volo!

            reverbGain = audioCtx.createGain();


            // Cablaggio Audio: Osc -> Filtro -> Delay & Reverb -> Master

            delayNode.connect(delayFeedback);

            delayFeedback.connect(delayNode);


            reverbNode.connect(reverbGain);


            document.getElementById('unlocker').style.transform = 'translateY(-110%)';

            updateFX();

        }


        // Funzione magica per generare un riverbero sintetico senza file esterni

        function createReverbImpulseResponse(duration, decay) {

            let sampleRate = 44100;

            let length = sampleRate * duration;

            let impulse = (window.AudioContext || window.webkitAudioContext) ? new AudioContext().createBuffer(2, length, sampleRate) : new webkitAudioContext().createBuffer(2, length, sampleRate);

            let left = impulse.getChannelData(0);

            let right = impulse.getChannelData(1);

            for (let i = 0; i < length; i++) {

                let percent = i / length;

                let formula = Math.pow(1 - percent, decay) * (Math.random() * 2 - 1);

                left[i] = formula; right[i] = formula;

            }

            return impulse;

        }


        function updateFX() {

            if(!audioCtx) return;

            let f = document.getElementById('filterKnob').value;

            let d = document.getElementById('delayKnob').value / 100;

            let r = document.getElementById('reverbKnob').value / 100;


            mainFilter.frequency.setValueAtTime(f, audioCtx.currentTime);

            delayFeedback.gain.setValueAtTime(d, audioCtx.currentTime);

            reverbGain.gain.setValueAtTime(r, audioCtx.currentTime);

        }


        function togglePlay() {

            if(!audioCtx) return;

            isPlaying = !isPlaying;

            let btn = document.getElementById('playBtn');

            if(isPlaying) {

                btn.innerText = "FERMA IL GIOCO 🛑";

                btn.classList.add('active');

                currentStep = 0;

                runLoop();

            } else {

                btn.innerText = "AVVIA MUSICA 🔄";

                btn.classList.remove('active');

                clearTimeout(timerId);

                document.querySelectorAll('.kid-bubble').forEach(b => b.classList.remove('playing'));

            }

        }


        function runLoop() {

            let bubbles = document.querySelectorAll('.kid-bubble');

            bubbles.forEach(b => b.classList.remove('playing'));

            

            let activeBubble = document.getElementById(`bubble-${currentStep}`);

            activeBubble.classList.add('playing');


            let stepData = stepsData[currentStep];

            if(stepData.active) {

                playFatTone(pentatonicScale[stepData.pitchStep], stepData.duration);

            }


            currentStep = (currentStep + 1) % 8;

            timerId = setTimeout(runLoop, 300); // Velocità fissa da filastrocca per bambini

        }


        function playFatTone(freq, duration) {

            let osc = audioCtx.createOscillator();

            let gain = audioCtx.createGain();


            osc.type = currentWave;

            osc.frequency.setValueAtTime(freq, audioCtx.currentTime);

            

            // Gestione dell'inviluppo della nota (attacco e rilascio morbido)

            gain.gain.setValueAtTime(0.25, audioCtx.currentTime);

            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);


            // Connessioni

            osc.connect(mainFilter);

            mainFilter.connect(gain);

            

            gain.connect(audioCtx.destination); // Suono Diretto

            gain.connect(delayNode);            // Mandata al Delay

            gain.connect(reverbNode);           // Mandata al Riverbero


            osc.start();

            osc.stop(audioCtx.currentTime + duration);

        }


        function playPreviewTone(freq, duration) {

            let osc = audioCtx.createOscillator();

            let gain = audioCtx.createGain();

            osc.type = currentWave;

            osc.frequency.setValueAtTime(freq, audioCtx.currentTime);

            gain.gain.setValueAtTime(0.15, audioCtx.currentTime);

            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);

            osc.connect(audioCtx.destination);

            osc.start(); osc.stop(audioCtx.currentTime + duration);

        }


        function resetAll() {

            stepsData = Array.from({length: 8}, () => ({ active: false, pitchStep: 0, duration: 0.2 }));

            document.querySelectorAll('.len-slider').forEach(s => s.value = 0.2);

            renderSequencerUI();

            document.getElementById('display').innerText = "Tutto pulito! Ricomincia da capo.";

        }
